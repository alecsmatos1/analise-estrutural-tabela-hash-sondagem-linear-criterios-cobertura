def chaves(self):
    """
        Devolve uma lista com todas as chaves presentes na tabela.
        """
    resultado = []
    for entrada in self.tabela:
        if entrada is not None and entrada is not REMOVIDO:
            resultado.append(entrada[0])
    return resultado
